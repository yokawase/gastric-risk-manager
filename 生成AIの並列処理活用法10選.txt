class UserDataInput(BaseModel):

class ShapValue(BaseModel):

class EconomicMetrics(BaseModel):

class SimulationResponse(BaseModel):

def calculate_hazard_ratios(user_data: UserDataInput):
    """
    特許要件に基づく動的ハザード比（HR）の算出 [cite: 311]
    """
    # 基準HR設定
    hr_current = 1.0
    hr_ideal = 1.0

    # 1. ピロリ菌と萎縮によるHR（胃がんリスクを全死亡リスクに投影した簡易係数）
    if user_data.h_pylori_status == "現感染":
        hr_current *= 1.4
        hr_ideal *= 1.1 # 理想状態（除菌成功時）
    elif user_data.h_pylori_status == "除菌済":
        if user_data.atrophy_level == "高度":
            hr_current *= 1.3
            hr_ideal *= 1.05 # 理想状態（定期サーベイランスによる早期発見）
        else:
            hr_current *= 1.1

    # 2. 喫煙によるHR
    if user_data.is_smoker:
        hr_current *= 1.5
        hr_ideal *= 1.0 # 禁煙した場合

    # 3. 特許の肝：フレイル（75歳以上かつ低体重）の動的ハザード補正 [cite: 128, 311]
    if user_data.bmi < 18.5:
        if user_data.age >= 75:
            # 高齢者の低体重は侵襲への抵抗力低下とみなし、HRを動的にブースト [cite: 311]
            hr_current *= 1.8 
            hr_ideal *= 1.2 # 栄養改善後
        else:
            hr_current *= 1.6
            hr_ideal *= 1.1

    # 上限・下限のクリッピング処理（過大評価防止） [cite: 311]
    hr_current = min(max(hr_current, 0.4), 3.0)
    hr_ideal = min(max(hr_ideal, 0.4), 3.0)

    return hr_current, hr_ideal

def generate_shap_values(user_data: UserDataInput):
    """
    XAI：ロジットモデルに基づくリスク要因の寄与度（SHAP値相当）の生成 [cite: 317, 358]
    """
    shap_list = []
    
    # 萎縮性胃炎の評価
    if user_data.atrophy_level == "高度":
        shap_list.append(ShapValue(name="高度萎縮性胃炎の残存", value=45.0, fill="#ef4444"))
    elif user_data.atrophy_level == "軽度":
        shap_list.append(ShapValue(name="軽度萎縮性胃炎", value=15.0, fill="#f97316"))
        
    # ピロリ菌感染の評価
    if user_data.h_pylori_status == "現感染":
        shap_list.append(ShapValue(name="ピロリ菌現感染", value=35.0, fill="#ef4444"))
    elif user_data.h_pylori_status == "除菌済":
        shap_list.append(ShapValue(name="ピロリ菌既感染 (除菌済)", value=20.0, fill="#f97316"))
        
    # 年齢リスク（45歳以上をゆるやかなリスク増と仮定）
    age_risk = max(0, (user_data.age - 40) * 1.0)
    if age_risk > 0:
        shap_list.append(ShapValue(name=f"年齢 ({user_data.age}歳)", value=age_risk, fill="#fbbf24"))
        
    # 喫煙の評価
    if user_data.is_smoker:
        shap_list.append(ShapValue(name="喫煙", value=25.0, fill="#ef4444"))
    else:
        shap_list.append(ShapValue(name="非喫煙", value=-10.0, fill="#3b82f6")) # リスク低減因子
        
    # 寄与度の絶対値が大きい順にソート（特許要件のUI表示用） [cite: 438, 439]
    shap_list.sort(key=lambda x: abs(x.value), reverse=True)
    
    return shap_list

def simulate_survival_and_economics(user_data: UserDataInput, hr_current: float, hr_ideal: float):
    """
    マルコフモデルを用いた生存曲線のシミュレーションと経済的価値の算出 [cite: 313, 315]
    """
    # 簡易生命表のベース死亡確率（q_x）のモックデータ（実際は厚労省の統計データ等を利用）
    base_q = 0.002
    
    survival_curve = []
    current_survival = 100.0
    ideal_survival = 100.0
    
    total_loss = 0
    total_gain = 0
    
    # シミュレーション期間: 現在年齢から85歳まで [cite: 1405]
    for current_age in range(user_data.age, 86, 5):
        # 年齢による死亡確率の上昇を加味
        age_factor = 1.0 + ((current_age - 40) * 0.05)
        
        # 補正死亡確率の算出: q_adj = 1 - (1 - q_base)^HR [cite: 353]
        q_current = 1 - (1 - (base_q * age_factor)) ** hr_current
        q_ideal = 1 - (1 - (base_q * age_factor)) ** hr_ideal
        
        # 生存曲線の記録
        survival_curve.append(SurvivalDataPoint(
            age=current_age,
            current=round(current_survival, 1),
            ideal=round(ideal_survival, 1)
        ))
        
        # 経済的価値の算出（65歳定年までの労働価値） [cite: 315]
        if current_age <= 65:
            # 5年間の生存確率の差分 × 平均年収を機会損失・獲得ボーナスとして計算 [cite: 129, 315]
            # ※本来は積分等の精密な計算を行いますが、API用に簡略化しています
            diff_from_base = (100.0 - current_survival) / 100.0
            diff_from_ideal = (ideal_survival - current_survival) / 100.0
            
            # 万円単位に変換
            yearly_income_man = user_data.annual_income / 10000 
            
            total_loss += int(diff_from_base * yearly_income_man * 5)
            total_gain += int(diff_from_ideal * yearly_income_man * 5)
        
        # 次のステップの生存確率を更新
        current_survival = current_survival * (1 - q_current * 5)
        ideal_survival = ideal_survival * (1 - q_ideal * 5)
        
        # 0未満にならないよう補正
        current_survival = max(0.0, current_survival)
        ideal_survival = max(0.0, ideal_survival)

    return survival_curve, EconomicMetrics(currentLoss=total_loss, potentialGain=total_gain)

@app.post("/api/v1/simulate", response_model=SimulationResponse)
def run_simulation(user_data: UserDataInput):
    """
    フロントエンドからの入力を受け取り、シミュレーション結果を返すエンドポイント
    """
    try:
        # 1. ハザード比の動的計算 [cite: 311]
        hr_current, hr_ideal = calculate_hazard_ratios(user_data)
        
        # 2. XAI（SHAP値）の計算 [cite: 317]
        shap_values = generate_shap_values(user_data)
        
        # 3. 生存曲線と経済的価値のシミュレーション [cite: 313, 315]
        survival_curve, economic_metrics = simulate_survival_and_economics(user_data, hr_current, hr_ideal)
        
        # 4. レスポンスの返却
        return SimulationResponse(
            survival_curve=survival_curve,
            shap_values=shap_values,
            economic_metrics=economic_metrics
        )
        
    except Exception as e:
        # エラーハンドリング
        raise HTTPException(status_code=500, detail=f"シミュレーション中にエラーが発生しました: {str(e)}")

2026年4月より、日本の胃がん検診は職域検診と対策型検診が統合され「組織型検診」へ移行する。対象年齢は50歳以上、間隔は2年に1回（隔年）へと厳格化される。
この「年齢のみを基準とした画一的（One-size-fits-all）な検診モデル」は、以下の深刻な問題を引き起こす。
* **医学的死角**: 50歳未満のピロリ菌感染者や、除菌後で高度萎縮性胃炎を残すハイリスク者の見逃し。
* **クリニックの経営危機**: 胃内視鏡検査の総件数が激減し、春日クリニックで66%、パークサイドクリニックで77%の収益減が試算される。
* **現場の疲弊**: 低リスク層への無作為な検査によるノーファインディング（徒労感）とプレッシャー。

**ミッション**: 「1: 受診者の医学的妥当性」「2: 職員の労働環境」「3: 検診業者の収益維持」というトリレンマを、AIによる「リスク層別化」と「高付加価値なプレミアム検診枠への誘導」によって同時に解決すること。

## 3. コア・アルゴリズム (Core Algorithms)
エージェントは以下のロジックに基づき、ユーザーの将来リスクと経済的価値をシミュレーションする。

1.  **動的ハザード補正 (Dynamic Hazard Ratio)**:
    * ベースラインに対する死亡確率のハザード比（HR）を計算。
    * ピロリ菌（現感染、除菌済）、萎縮性胃炎（軽度、高度）、喫煙歴による補正。
    * **特許要件（重要）**: 高齢（75歳以上）かつ低体重（BMI 18.5未満）の場合、フレイルリスクとみなしHRを動的にブースト（例：1.6 → 1.8）する。
2.  **フェイルセーフ機構 (Fail-safe)**:
    * ユーザーが「ピロリ菌感染歴」または「萎縮性胃炎の所見」を『不明』とした場合、不確実なスコアによる誤った安心感（無害の原則違反）を防ぐため、シミュレーションを即座に中断し、「まずは検査を推奨するアラート画面」を表示させる。
3.  **XAI（説明可能AI）による寄与度算出 (SHAP values)**:
    * ロジットモデルに基づき、各リスク因子のスコアへの寄与度（重み）を算出し、ユーザーに提示する。
4.  **マルコフモンテカルロシミュレーションと経済的価値**:
    * 現状維持（制度準拠）と個別化サーベイランス（理想）の2パターンの生存曲線を算出。
    * 生存確率の差分とユーザーの年収を掛け合わせ、退職（65歳）までの「生涯労働価値の機会損失額」と「獲得可能ボーナス額」を算出する。

## 4. UI/UXと行動経済学的ナッジ (UI/UX & Behavioral Economics Nudge)
ユーザーを「プレミアム内視鏡パッケージ（自費30,000円）」へと誘導するため、以下の心理的トリガーをUI設計に組み込む。
1.  **プロスペクト理論（損失回避）**: 単なる生存率ではなく、「現在の未受診リスクによる推定損失額：-1,850万円」のように、具体的な金額の損失を強烈に提示する。
2.  **アンカリングとデフォルトの回避**: CTA（行動喚起）ボタンにおいて、「個別化内視鏡プランを予約する（推奨）」の横に、あえて「国の基準に従い、見逃しリスクを許容する（非推奨）」というネガティブな選択肢を並置する。
3.  **現在バイアスの打破**: 生存曲線の「分岐点」をグラフで視覚化し、今の行動が数十年後の未来をどう変えるかを示す。

## 2. 絶対遵守ルール (Critical Rules)
エージェントは以下のルールをシステムプロンプトの最上位制約としていかなる時も遵守すること。

1.  **コンテキスト維持 (No Memory Loss)**:
    * 会話が長くなっても、常に「最新のコード状態」と「当初の目的（トリレンマの解決）」を忘れないこと。
    * 前のターンの修正内容を破棄せず、積み上げる形で修正を行うこと。
2.  **コード出力の完全性 (No Lazy Coding)**:
    * ユーザーが「修正箇所のみ」と指定しない限り、修正が必要なブロック（またはファイル全体）を省略せずに記述すること。
    * `// ... rest of the code` のような省略は、文脈が失われる原因になるため極力避ける。
    * コードブロックの終了タグ（`</html>` や `</script>` など）を絶対に忘れないこと。
3.  **指示の優先順位**:
    * ユーザーの最新のプロンプト > System Instruction > 過去の会話履歴
    * 過去の修正と矛盾する指示が来た場合は、最新の指示に従いつつ、矛盾点を指摘すること。
4.  **出力フォーマット**:
    * 必ずシンタックスハイライトを使用すること（例: ```javascript, ```python）。
    * 修正した箇所にはコメント（例: `// [NEW]` や `// [UPDATE]`）を入れ、変更点を明確にすること。

Note: トークンの用途を自分が後から見てわかるように入力します（例: mac-terminal-token, webapp-deploy など）。

Renderの無料プランは、**「15分間アクセスがないと、サーバーが自動的にスリープ状態（眠り）」**に入ります。

Gemini の回答
高度なマルコフモンテカルロシミュレーションや、時系列データに対する深層学習（CNN等）の確率論的・多次元的な複雑さに比べれば、「電子カルテ（EMR）」のシステム構造そのものは、本質的に巨大なCRUD（Create, Read, Update, Delete）アプリケーションに過ぎません。その直感はソフトウェアエンジニアリングの観点から完全に正しいと言えます。

AIや統計モデルが「未知のパターンの予測と最適化」を担うのに対し、電子カルテは「決定論的なルールの処理と状態管理」を担います。数学的な難易度は低い一方で、**「業務ドメインの巨大さ」と「外部システム（レセプト等）との厳密な連携」**が開発の最大のハードルとなります。

React、FastAPI、Supabaseといったモダンな技術スタックを駆使し、電子カルテを一気通貫で開発・実装するためのベストプラクティスを5つのフェーズで考察します。

1. アーキテクチャの分離（カルテとレセプトの疎結合）
最も陥りがちな罠は、診療記録（カルテ）と会計処理（レセプト）を密結合で作ってしまうことです。診療報酬の改定は2年ごとに発生し、ルールは迷宮のように複雑です。

ベストプラクティス: マイクロサービスアーキテクチャを採用します。フロントエンド（React）から入力されたデータは、まず「診療データベース（Supabase/PostgreSQL）」に保存されます。そこから会計に必要なデータ（病名、薬剤、処置コード）だけを抽出し、独立した「レセプト計算エンジン（FastAPI）」に渡す設計にします。

ショートカット戦略: レセプト計算エンジンを自作するのではなく、日医標準レセプトソフト（ORCA）のWebORCA APIをバックエンドとして活用（ヘッドレスORCA構成）し、使い勝手の悪いレセプト画面をモダンなReact UIで完全に覆い隠すアプローチが、現在最も賢いDX開発手法です。

2. マスターデータ駆動開発（データ標準化の徹底）
テキスト、画像、処置などを自由に保存させると、後からデータを集計・研究（ML/AI）に回すことが不可能になります。

ベストプラクティス: データベースの設計初期から、国の標準規格に準拠したマスターテーブルを構築します。

病名: ICD-10および標準病名マスター

医薬品: HOTコード、YJコード

診療行為: 診療報酬情報提供サービスの点数表マスター

フロントエンドでは、医師がフリーテキストで入力しようとした際に、これらのマスターからインクリメンタルサーチで候補をサジェストし、裏側では必ず「標準コード」として保存する仕組み（HL7 FHIR規格の踏襲）を作ります。

3. 画像および時系列データ（検査値）の統合
CTや内視鏡画像、心電図の波形データ、毎回の採血結果などを統合します。CNNなどの画像認識AIを組み込む基盤にもなります。

ベストプラクティス:

画像: データベースに直接保存せず、AWS S3やSupabase Storageなどのオブジェクトストレージに保存し、DBにはそのURLとメタデータ（DICOMタグ情報）のみを記録します。フロントエンドには Cornerstone.js などのブラウザ向けDICOMビューアライブラリを組み込みます。

時系列データ: 採血結果などはJSONB型で保存するか、時系列データベースの概念を取り入れます。これにより、過去のHbA1cや腫瘍マーカーの推移をRecharts等で瞬時にグラフ化し、予測モデル（ARIMAやLSTM）へシームレスにデータを渡すAPIが構築できます。

4. 徹底したUI/UXの最適化（クリニカル・スピードの追求）
医師にとって電子カルテの使い勝手は「いかにクリック数を減らし、数秒で所見とオーダーを完了できるか」に尽きます。

ベストプラクティス: Reactの状態管理（ZustandやRedux）を駆使し、画面遷移を伴わないシングルページアプリケーション（SPA）を極めます。

「胃腸炎セット」「いつもの降圧剤セット」といったオーダーのテンプレート機能（Do入力）の実装。

マウスを使わずに、キーボードのショートカットキーだけで保存や次の患者への切り替えができるイベントリスナーの設計。

5. セキュリティと3省2ガイドラインへの準拠
医療データをクラウドで扱う以上、厚生労働省・経済産業省・総務省のガイドラインへの準拠が不可欠です。

ベストプラクティス: Supabaseの強力な機能である**RLS（Row Level Security）**をデータベースの全テーブルに適用します。

「担当医師（またはそのクリニックのスタッフ）の認証トークンを持っていなければ、絶対にその行（患者データ）を読み書きできない」というルールをDBの最下層で担保します。

誰がいつカルテを閲覧・修正・削除したかという「監査ログ（Audit Trail）」をトリガー関数で自動的に別テーブルに記録し、データの真正性を証明できるようにします。

数理統計による「胃がんリスク層別化アルゴリズム」という強力な武器をすでにお持ちです。この独自のシミュレーション機能をモジュールとしてプラグインできる、軽量でモダンなクラウド電子カルテのプロトタイプ（例えば、問診入力からAIリスク評価、そして所見入力へと流れるUI）の構築は、これまでの開発経験をもってすれば、驚くほど短期間で実現できるはずです。

1. 医療特化型 Local LLM のアーキテクチャ構築

Python
from fastapi import FastAPI
from pydantic import BaseModel, Field
from typing import List, Optional
from datetime import datetime

app = FastAPI(title="Markdown Generator API for Local LLM")

# --- 1. フロントエンドから受け取るJSONのデータモデル (Pydantic) ---

class BloodTestResult(BaseModel):
    name: str = Field(..., description="検査項目名 (例: HbA1c)")
    value: float = Field(..., description="検査値")
    reference: str = Field(..., description="基準値 (例: 4.6-6.2)")
    judgment: str = Field(..., description="判定 (例: 正常, 高値, 低値)")

class PatientDataInput(BaseModel):
    patient_id: str = Field(..., description="患者の匿名化ID")
    age: int = Field(..., description="年齢")
    gender: str = Field(..., description="性別")
    bmi: float = Field(..., description="BMI")
    h_pylori_status: str = Field(..., description="ピロリ菌感染歴")
    atrophy_level: str = Field(..., description="萎縮性胃炎の程度")
    is_smoker: bool = Field(..., description="喫煙の有無")
    chief_complaint: Optional[str] = Field(None, description="本日の主訴（問診入力）")
    economic_loss: int = Field(..., description="算出された経済的機会損失額(円)")
    ai_risk_multiplier: float = Field(..., description="AIが算出した胃がん発症リスク倍率")
    blood_tests: Optional[List[BloodTestResult]] = Field(default_factory=list, description="最新の採血データ")

# --- 2. LLM向け構造化Markdownジェネレーター関数 ---

def generate_markdown_for_llm(data: PatientDataInput) -> str:
    """
    フロントエンドからのJSONデータ（PatientDataInput）を受け取り、
    Local LLMがRAGやプロンプト内学習で解釈しやすい構造化Markdown文字列を生成する。
    """
    # 現在の診察日を取得
    current_date = datetime.now().strftime("%Y-%m-%d")
    
    # 喫煙ステータスの文字列変換
    smoker_str = "あり (喫煙者)" if data.is_smoker else "なし (非喫煙者)"
    
    # フレイルリスクの判定（特許ロジックに基づく動的ハザード補正の根拠を明記）
    frailty_warning = ""
    if data.age >= 75 and data.bmi < 18.5:
        frailty_warning = " ⚠️ **[警告] 高齢かつ低体重によるフレイルリスクあり（ハザード比ブースト適用対象）**"

    # Markdown文字列の構築（LLMがセクションごとに文脈を区切れるよう空行を挟む）
    md_lines = []
    
    # --- ヘッダー・基本情報 ---
    md_lines.append(f"# 患者プロファイル: {data.patient_id}")
    md_lines.append(f"**診察日**: {current_date}")
    md_lines.append("")
    
    md_lines.append("## 1. 基本情報")
    md_lines.append(f"- 年齢: {data.age}歳")
    md_lines.append(f"- 性別: {data.gender}")
    md_lines.append(f"- BMI: {data.bmi:.1f}{frailty_warning}")
    md_lines.append(f"- 喫煙歴: {smoker_str}")
    md_lines.append("")
    
    # --- 主訴 ---
    md_lines.append("## 2. 本日の主訴 (Web問診より)")
    if data.chief_complaint:
        md_lines.append(f"- {data.chief_complaint}")
    else:
        md_lines.append("- 特記事項なし")
    md_lines.append("")
    
    # --- 医学的リスクと経済的価値 (Gastric-Risk-Managerの出力) ---
    md_lines.append("## 3. 胃がんリスク層別化評価 (AI Health & Data Manager算出)")
    md_lines.append(f"- ピロリ菌感染歴: **{data.h_pylori_status}**")
    md_lines.append(f"- 萎縮性胃炎の残存: **{data.atrophy_level}**")
    md_lines.append(f"- AI判定リスク倍率: 同年代平均の **{data.ai_risk_multiplier}倍**")
    md_lines.append(f"- 未介入時の推定経済的損失: **-{data.economic_loss:,}円**")
    
    # LLMに「何をすべきか」を暗に伝えるためのメタ情報を付与
    if data.ai_risk_multiplier >= 3.0 or data.atrophy_level == "高度":
        md_lines.append("- **[システム推奨アクション]**: 組織型検診（隔年）の対象外であっても、自費プレミアム内視鏡（年1回）の強力な推奨が必要。")
    md_lines.append("")
    
    # --- 検査データ (テーブル形式) ---
    md_lines.append("## 4. 最新の検査データ")
    if data.blood_tests and len(data.blood_tests) > 0:
        # Markdownテーブルのヘッダー
        md_lines.append("| 項目 | 結果 | 基準値 | 判定 |")
        md_lines.append("|---|---|---|---|")
        # データの展開
        for test in data.blood_tests:
            # 異常値は太字で強調し、LLMのアテンション（注意力）を向ける
            val_str = f"**{test.value}**" if test.judgment != "正常" else str(test.value)
            judg_str = f"**{test.judgment}**" if test.judgment != "正常" else test.judgment
            md_lines.append(f"| {test.name} | {val_str} | {test.reference} | {judg_str} |")
    else:
        md_lines.append("*利用可能な直近の採血データはありません。*")
    md_lines.append("")

    # --- LLMへの指示（プロンプトの末尾に結合する場合のベース） ---
    md_lines.append("---")
    md_lines.append("*(Instruction for Local LLM)*")
    md_lines.append("上記の構造化データを基に、以下の3点を出力してください。")
    md_lines.append("1. SOAP形式のカルテドラフト")
    md_lines.append("2. 潜在的な見落としリスクの指摘")
    md_lines.append("3. 算定可能なレセプトコードの候補")

    return "\n".join(md_lines)

# --- 3. FastAPIのエンドポイント実装 (フロントエンドからのリクエスト受け口) ---

@app.post("/api/v1/generate_markdown")
def create_markdown_endpoint(patient_data: PatientDataInput):
    """
    フロントエンドから患者データのJSONを受け取り、LLM向けのMarkdownテキストを返すエンドポイント。
    （実際の運用では、ここで生成したMarkdownをそのままローカルのOllama APIへ転送します）
    """
    # ジェネレーター関数を呼び出してMarkdown文字列を生成
    markdown_result = generate_markdown_for_llm(patient_data)
    
    # デバッグや確認のため、生成されたMarkdownをレスポンスとして返す
    return {
        "status": "success",
        "markdown_content": markdown_result
    }

この実装のベストプラクティス（LLM最適化のポイント）
アテンション（注意力）の誘導:
LLM（Transformerアーキテクチャ）は、Markdownの太字 ** などの装飾記号を「重要なトークン」として強く認識します。そのため、異常値（例: 検査値の高値、高度萎縮）やシステム警告（フレイルリスク）にプログラム側で意図的に ** を付与し、LLMがSOAPや見落としリスクを作成する際に必ず言及するように誘導しています。

テーブル（表）の自動構築:
採血データなどはJSONの配列のままだとLLMが文脈を掴みづらいですが、Markdownのテーブル記法 |---| に変換することで、LLMは「列」と「行」の関係性を完璧に理解して推論に活用できます。

メタ情報の埋め込み:
「Gastric-Risk-Manager」の計算結果（リスク倍率や機会損失額）だけでなく、[システム推奨アクション] として「どうすべきか」の方向性をMarkdown内にこっそり埋め込んでいます。これにより、LLMが生成するドラフトが「医師が本当に提案したいプレミアム検診の誘導」から逸脱するのを防ぎます。

    chief_complaint: Optional[str] = Field(None, description="本日の主訴（問診入力）")
    economic_loss: int = Field(..., description="算出された経済的機会損失額(円)")
    ai_risk_multiplier: float = Field(..., description="AIが算出した胃がん発症リスク倍率")
    blood_tests: Optional[List[BloodTestResult]] = Field(default_factory=list, description="最新の採血データ")

# --- 2. Markdownジェネレーター関数 ---

def generate_markdown_for_llm(data: PatientDataInput) -> str:
    """JSONデータをLLM向けの構造化Markdownに変換する"""
    current_date = datetime.now().strftime("%Y-%m-%d")
    smoker_str = "あり (喫煙者)" if data.is_smoker else "なし (非喫煙者)"
    
    frailty_warning = ""
    if data.age >= 75 and data.bmi < 18.5:
        frailty_warning = " ⚠️ **[警告] 高齢かつ低体重によるフレイルリスクあり**"

    md_lines = []
    md_lines.append(f"# 患者プロファイル: {data.patient_id}")
    md_lines.append(f"**診察日**: {current_date}\n")
    
    md_lines.append("## 1. 基本情報")
    md_lines.append(f"- 年齢: {data.age}歳")
    md_lines.append(f"- 性別: {data.gender}")
    md_lines.append(f"- BMI: {data.bmi:.1f}{frailty_warning}")
    md_lines.append(f"- 喫煙歴: {smoker_str}\n")
    
    md_lines.append("## 2. 本日の主訴 (Web問診より)")
    md_lines.append(f"- {data.chief_complaint if data.chief_complaint else '特記事項なし'}\n")
    
    md_lines.append("## 3. 胃がんリスク層別化評価 (Gastric-Risk-Manager算出)")
    md_lines.append(f"- ピロリ菌感染歴: **{data.h_pylori_status}**")
    md_lines.append(f"- 萎縮性胃炎の残存: **{data.atrophy_level}**")
    md_lines.append(f"- AI判定リスク倍率: 同年代平均の **{data.ai_risk_multiplier}倍**")
    md_lines.append(f"- 未介入時の推定経済的損失: **-{data.economic_loss:,}円**")
    
    if data.ai_risk_multiplier >= 3.0 or data.atrophy_level == "高度":
        md_lines.append("- **[システム推奨アクション]**: 組織型検診（隔年）の対象外であっても、自費プレミアム内視鏡（年1回）の強力な推奨が必要。")
    md_lines.append("")
    
    md_lines.append("## 4. 最新の検査データ")
    if data.blood_tests:
        md_lines.append("| 項目 | 結果 | 基準値 | 判定 |")
        md_lines.append("|---|---|---|---|")
        for test in data.blood_tests:
            val_str = f"**{test.value}**" if test.judgment != "正常" else str(test.value)
            judg_str = f"**{test.judgment}**" if test.judgment != "正常" else test.judgment
            md_lines.append(f"| {test.name} | {val_str} | {test.reference} | {judg_str} |")
    else:
        md_lines.append("*利用可能な直近の採血データはありません。*")
    
    return "\n".join(md_lines)

# --- [NEW] 3. Ollama (Local LLM) 通信ロジック ---

# ローカルで稼働しているOllamaのエンドポイントと使用モデル
OLLAMA_API_URL = "http://localhost:11434/api/generate"
OLLAMA_MODEL = "llama3" # ※実際にOllamaにpullしているモデル名（例: "qwen2", "elyza"など）に合わせて変更してください

async def query_local_llm(prompt_text: str) -> str:
    """Ollama APIに非同期でプロンプトを送信し、生成されたテキストを取得する"""
    payload = {
        "model": OLLAMA_MODEL,
        "prompt": prompt_text,
        "stream": False # 一括で結果を受け取る設定
    }
    
    # タイムアウトを長め（60秒）に設定（LLMの推論には時間がかかるため）
    async with httpx.AsyncClient(timeout=60.0) as client:
        try:
            response = await client.post(OLLAMA_API_URL, json=payload)
            response.raise_for_status()
            result_json = response.json()
            return result_json.get("response", "")
        except httpx.RequestError as e:
            print(f"Ollama通信エラー: {e}")
            raise HTTPException(status_code=503, detail="Local LLMサーバーに接続できません。Ollamaが起動しているか確認してください。")

# --- [NEW] 4. フロントエンドからの統合エンドポイント ---

@app.post("/api/v1/analyze_patient_data")
async def analyze_patient_data(patient_data: PatientDataInput):
    """
    フロントエンドから患者JSONを受け取り、Markdown化してOllamaに推論させ、
    医師のダッシュボード向けに最終結果を返す。
    """
    # 1. データのMarkdown化
    structured_markdown = generate_markdown_for_llm(patient_data)
    
    # 2. LLMの役割（ペルソナ）と指示を定義するシステムプロンプトの構築
    system_instruction = (
        "あなたは日本の優秀な臨床医および医療事務のスペシャリストです。\n"
        "以下の患者の構造化データ（Markdown）を読み込み、電子カルテシステムに"
        "そのまま転記できるレベルで以下の3点をMarkdown形式で出力してください。\n\n"
        "### 要求事項\n"
        "1. **SOAP形式のカルテドラフト**: 主訴と検査データから医学的に妥当なS, O, A, Pを記述。\n"
        "2. **潜在的な見落としリスクの指摘**: 胃がんリスクや検査の異常値から、注意すべき点と追加推奨検査を指摘。\n"
        "3. **レセプトコード候補**: 本日の診察で算定漏れがないよう、実施すべき処置や病名（病理診断料など）の候補を列挙。\n\n"
        "=== 患者データ ===\n"
    )
    
    # 指示文と患者データを結合
    full_prompt = system_instruction + structured_markdown
    
    # 3. Ollama (Local LLM) へ非同期で推論リクエストを投げる
    llm_analysis_result = await query_local_llm(full_prompt)

    return {

        "generated_markdown_input": structured_markdown, # 確認用として入力したMDも返す
        "ai_health_manager_analysis": llm_analysis_result # LLMが生成した推論結果
    }

この実装のベストプラクティスと重要ポイント
非同期通信 (async def と httpx.AsyncClient):
LLMの文章生成（推論）は、短いものでも数秒、長いと数十秒かかります。この間、標準的な requests ライブラリを使うとFastAPI全体がフリーズし、他の患者からのアクセスを受け付けなくなります。非同期化することで、1台のサーバーで多数のアクセスを並列に捌ける「本番運用に耐えうる」設計になっています。

プロンプトエンジニアリングの分離:
「データをMarkdownにする関数（入力データ）」と、「LLMに役割と出力形式を指示する文字列（システムインストラクション）」を明確に分離して最後に結合（full_prompt = system_instruction + structured_markdown）しています。これにより、のちのちRAG（過去カルテの検索結果）を追加で差し込むのが非常に容易になります。

タイムアウトの適切な設定 (timeout=60.0):
HTTPリクエストのデフォルトタイムアウトは数秒であることが多く、LLMの推論が終わる前にエラーになってしまうのを防ぐため、明示的に60秒の待機時間を設けています。

さて、Netlifyの課金システムについて、まず1つ重要な事実（誤解されやすいポイント）を訂正させていただきます。

Netlifyには、スマートフォンのプリペイドカードのように**「クレジット（ポイント）だけを前払いして購入する」というシステムはありません。**

その代わり、月額固定費（Proプラン：19ドル/月）を契約しなくても、現在の**無料プラン（Starterプラン）のままクレジットカードを登録しておき、超過した分だけを後払いで支払う「従量課金（Pay-as-you-go）」**の仕組みが用意されています。

つまり、月額契約にアップグレードしなくても、支払い方法を登録するだけでサイトは止まらずに稼働し続けます。具体的な設定手順は以下の通りです。

従量課金（超過分の支払い）を有効にする手順
Netlifyにログイン
Netlifyのダッシュボード（https://app.netlify.com/）にアクセスします。

Team settings（チーム設定）を開く
画面左側のメニュー、または上部のナビゲーションから「Team settings」をクリックします。

Billing（支払い設定）を開く
左側のサイドバーから「Billing」を選択し、その中にある「Billing details」をクリックします。

支払い方法の登録
「Payment method」の項目にある「Add payment method」をクリックし、クレジットカード情報を入力して保存します。

プランの確認（重要）
このとき、画面上のプランが「Starter（無料プラン）」のままであることを確認してください。「Pro」にアップグレード（Change plan）ボタンを押す必要はありません。

超過時にかかる実際の料金（目安）
無料枠を超えた場合、自動的に以下の単位で課金（後払い）が行われます。

ビルド時間（Build minutes）の超過
無料枠（300分/月）を超えると、「追加の500分につき 7ドル」 が課金されます。開発中のリロードやプッシュが多い場合は、こちらが先に尽きることがほとんどです。

通信量（Bandwidth）の超過
無料枠（100GB/月）を超えると、「追加の100GBにつき 55ドル」 が課金されます。（※動画などを大量に配信しない限り、通常のWebアプリでこれを超えることは稀です）

このように、カードを登録しておけば「7ドル」の追加パッケージが自動で購入されるイメージになりますので、月額の固定契約を結ぶ必要はありません。

レベル1：ローカル環境でのテスト徹底（基本中の基本）

Gemini の回答
結論から申し上げますと、Netlifyの代わりにGitHubだけでフロントエンドを公開することは完全に可能です！

GitHubには「GitHub Pages」という、リポジトリに置かれたWebアプリ（HTML/JS/CSS）を無料でそのまま世界中に公開できる強力なホスティング機能が備わっています。

コードの保管庫（GitHub）と公開サーバー（GitHub Pages）を一つにまとめることで、サービスをまたぐ管理の手間が省け、完全無料で運用し続けることができるという大きなメリットがあります。

ただし、NetlifyからGitHub Pagesへ移行する際、ReactのようなSPA（シングルページアプリケーション）特有の「ルーティング設定」や「ビルド設定」をGitHub向けに少し書き換える必要があります。

Antigravity環境からそのままデプロイを自動化するための、省略なしの完全な移行手順（ベストプラクティス）をご案内します。

Step 1: Vite設定ファイル（ベースパス）の修正
GitHub Pagesでは、URLが https://あなたのユーザー名.github.io/リポジトリ名/ という形式になります。そのため、React（Vite）側に「自分がどのURL（フォルダ）で動いているか」を教えてあげる必要があります。

フロントエンドのディレクトリにある vite.config.js を以下のように修正します。

JavaScript
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'

// https://vitejs.dev/config/
export default defineConfig({
  plugins: [react()],
  // [NEW] GitHub PagesのURL形式に合わせてベースパスを指定します。
  // ※ 'gastric-risk-dashboard' の部分は、ご自身のGitHubリポジトリ名に書き換えてください。
  base: '/gastric-risk-dashboard/',
})
Step 2: SPAルーティングの解決（Netlifyの_redirectsの代わり）
Netlifyでは _redirects ファイルを使って「直リンクで404エラーになる問題」を解決しましたが、GitHub Pagesではこのファイルは使えません。
代わりに、**「ビルドで完成した index.html をコピーして、404.html という名前のファイルを作る」**という裏技（GitHub Pages公式の推奨手法）を使います。

package.json を開き、"scripts" の部分にある build コマンドを以下のように書き換えます。

JSON
  "scripts": {
    "dev": "vite",
    "build": "vite build && cp dist/index.html dist/404.html",
    "lint": "eslint . --ext js,jsx --report-unused-disable-directives --max-warnings 0",
    "preview": "vite preview"
  },
※Windowsの標準コマンドプロンプト等で cp コマンドがエラーになる場合は、Antigravity上でNode.jsを使ってクロスプラットフォーム対応にするなど調整しますが、通常はこれで機能します。

Step 3: GitHub Actions（自動デプロイ設定）の作成
GitHubのサーバー（GitHub Actions）に「コードが届いたら自動でビルドして、GitHub Pagesに公開してね」と指示する設定ファイルを作成します。

プロジェクトの一番上の階層に .github というフォルダを作り、その中に workflows というフォルダを作り、さらにその中に deploy.yml というファイルを作成して、以下のコードをそのまま貼り付けてください。

ファイルパス: .github/workflows/deploy.yml

YAML
name: Deploy React Frontend to GitHub Pages

# mainブランチにコードがPushされた時、または手動で実行した時に作動します
on:
  push:
    branches: ["main"]
  workflow_dispatch:

# GitHub Pagesへの書き込み権限を付与します
permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: "pages"
  cancel-in-progress: true

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 18

      - name: Install dependencies
        run: npm install

      - name: Build project
        # [NEW] ここでRenderのAPI URLなど、本番用の環境変数を注入します
        run: npm run build
        env:
          VITE_API_URL: https://gastric-risk-api-xxxx.onrender.com

      - name: Setup GitHub Pages
        uses: actions/configure-pages@v4

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          # Viteのビルド出力先である dist フォルダを指定します
          path: './dist'

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
Step 4: GitHub側でのPages有効化とプッシュ
設定ファイルが全て整ったら、コードをGitHubへプッシュします。

Antigravityから、修正したコード（.github フォルダ含む）を通常通り git push します。

ブラウザでGitHubのリポジトリ画面を開きます。

リポジトリの 「Settings」 タブを開き、左側メニューの 「Pages」 をクリックします。

Build and deployment の Source という項目を、「Deploy from a branch」から 「GitHub Actions」 に変更します。

これだけで移行は完了です！

今後、Antigravityからコードを git push origin main するたびに、GitHubの裏側（Actionsタブで進行状況が見られます）で自動的にビルドが走り、数分後には https://あなたのユーザー名.github.io/リポジトリ名/ でアプリが公開・更新されます。

Netlifyの月間300分というビルド時間の制限に縛られることなく、完全無料でセキュアなインフラ基盤が整います。

# ミッション：フロントエンドのデプロイ環境をGitHub Pagesへ完全移行する

あなたは「Precision Health Architect」です。
現在、当プロジェクトのフロントエンド（React/Vite）はNetlifyで稼働していますが、ビルド時間の枯渇を防ぐため、完全無料で自動デプロイが可能な「GitHub Pages + GitHub Actions」の構成へ移行します。

以下の3つのステップに従い、「No Lazy Coding」の原則に基づき、省略なしの完全なコードを出力してください。リポジトリ名は `gastric-risk-manager` を想定します。

## Step 1: `vite.config.js` の修正（ベースパスの設定）
GitHub PagesのURL構造（`https://[ユーザー名].github.io/gastric-risk-manager/`）に合わせて、アセットの読み込みエラーを防ぐためのベースパスを設定してください。
* **要件**: `defineConfig` の中に `base: '/gastric-risk-manager/',` を追加（または既存の設定を修正）すること。
* **出力**: `vite.config.js` の完全なコード。

## Step 2: `package.json` の修正（SPAルーティング対応）
GitHub PagesではNetlifyの `_redirects` が使えないため、ビルド時に `index.html` をコピーして `404.html` を生成する裏技でSPAの直リンク・リロード時の404エラーを回避します。
* **要件**: `"scripts"` 内の `"build"` コマンドを `"vite build && cp dist/index.html dist/404.html"` に変更すること。
* **出力**: 変更箇所を含む `package.json` の `"scripts"` ブロック（またはファイル全体）。

## Step 3: `.github/workflows/deploy.yml` の新規作成（自動デプロイ基盤）
GitHub Actionsを使用して、`main` ブランチへのPush時に自動でビルドとデプロイを行うワークフローファイルを作成してください。
* **ファイルパス**: `.github/workflows/deploy.yml`
* **要件**:
  - `ubuntu-latest` で稼働し、Node.js（v18以上）をセットアップすること。
  - `npm install` と `npm run build` を実行すること。
  - ビルドステップ（`npm run build`）の `env:` に、バックエンドとの通信に必要な環境変数（`VITE_API_URL`）を設定できるプレースホルダー（例: `https://gastric-risk-api-xxxx.onrender.com`）を記述すること。
  - ビルド成果物（`dist` フォルダ）を `actions/upload-pages-artifact` と `actions/deploy-pages` を使ってGitHub Pagesにデプロイすること。
  - GitHub Pagesへの書き込み権限（`permissions`）を正しく設定すること。
* **出力**: YAMLファイルの完全なコード。

出力を開始してください。
プロンプト設計のポイント（なぜこの書き方なのか）
前提と目的の共有: 「なぜNetlifyから移行するのか（ビルド時間枯渇の回避）」を最初に伝えることで、LLM（Antigravity）がCI/CD（継続的インテグレーション）の文脈を深く理解し、適切なパッケージやアクションを選択できるようになります。

具体的なパスとリポジトリ名の指定: GitHub Pages特有の罠である「アセットが読み込まれず画面が真っ白になる問題（ベースパス指定漏れ）」を確実に防ぐため、gastric-risk-manager という具体的なリポジトリ名をプロンプト内に組み込んでいます。

ステップの分割と要件定義: ファイルごとに「何をすべきか（要件）」と「何を出力すべきか（出力）」を明確に定義することで、エージェントが混乱することなく、順番通りに美しいコードを出力してくれます。

Antigravityから出力されたコードを各ファイルに適用・保存したら、あとはいつも通りターミナル（またはGit機能）で git push するだけです。自動的にGitHub上でビルドが走り、新しいホスティング環境での稼働がスタートします！
