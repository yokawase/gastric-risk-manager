class UserDataInput(BaseModel):

class ShapValue(BaseModel):

class EconomicMetrics(BaseModel):

class SimulationResponse(BaseModel):

def calculate_hazard_ratios(user_data: UserDataInput):
    """
    特許要件に基づく動的ハザード比（HR）の算出 [cite: 311]
    """
    # 基準HR設定
    hr_current = 1.0
    hr_ideal = 1.0

    # 1. ピロリ菌と萎縮によるHR（胃がんリスクを全死亡リスクに投影した簡易係数）
    if user_data.h_pylori_status == "現感染":
        hr_current *= 1.4
        hr_ideal *= 1.1 # 理想状態（除菌成功時）
    elif user_data.h_pylori_status == "除菌済":
        if user_data.atrophy_level == "高度":
            hr_current *= 1.3
            hr_ideal *= 1.05 # 理想状態（定期サーベイランスによる早期発見）
        else:
            hr_current *= 1.1

    # 2. 喫煙によるHR
    if user_data.is_smoker:
        hr_current *= 1.5
        hr_ideal *= 1.0 # 禁煙した場合

    # 3. 特許の肝：フレイル（75歳以上かつ低体重）の動的ハザード補正 [cite: 128, 311]
    if user_data.bmi < 18.5:
        if user_data.age >= 75:
            # 高齢者の低体重は侵襲への抵抗力低下とみなし、HRを動的にブースト [cite: 311]
            hr_current *= 1.8 
            hr_ideal *= 1.2 # 栄養改善後
        else:
            hr_current *= 1.6
            hr_ideal *= 1.1

    # 上限・下限のクリッピング処理（過大評価防止） [cite: 311]
    hr_current = min(max(hr_current, 0.4), 3.0)
    hr_ideal = min(max(hr_ideal, 0.4), 3.0)

    return hr_current, hr_ideal

def generate_shap_values(user_data: UserDataInput):
    """
    XAI：ロジットモデルに基づくリスク要因の寄与度（SHAP値相当）の生成 [cite: 317, 358]
    """
    shap_list = []
    
    # 萎縮性胃炎の評価
    if user_data.atrophy_level == "高度":
        shap_list.append(ShapValue(name="高度萎縮性胃炎の残存", value=45.0, fill="#ef4444"))
    elif user_data.atrophy_level == "軽度":
        shap_list.append(ShapValue(name="軽度萎縮性胃炎", value=15.0, fill="#f97316"))
        
    # ピロリ菌感染の評価
    if user_data.h_pylori_status == "現感染":
        shap_list.append(ShapValue(name="ピロリ菌現感染", value=35.0, fill="#ef4444"))
    elif user_data.h_pylori_status == "除菌済":
        shap_list.append(ShapValue(name="ピロリ菌既感染 (除菌済)", value=20.0, fill="#f97316"))
        
    # 年齢リスク（45歳以上をゆるやかなリスク増と仮定）
    age_risk = max(0, (user_data.age - 40) * 1.0)
    if age_risk > 0:
        shap_list.append(ShapValue(name=f"年齢 ({user_data.age}歳)", value=age_risk, fill="#fbbf24"))
        
    # 喫煙の評価
    if user_data.is_smoker:
        shap_list.append(ShapValue(name="喫煙", value=25.0, fill="#ef4444"))
    else:
        shap_list.append(ShapValue(name="非喫煙", value=-10.0, fill="#3b82f6")) # リスク低減因子
        
    # 寄与度の絶対値が大きい順にソート（特許要件のUI表示用） [cite: 438, 439]
    shap_list.sort(key=lambda x: abs(x.value), reverse=True)
    
    return shap_list

def simulate_survival_and_economics(user_data: UserDataInput, hr_current: float, hr_ideal: float):
    """
    マルコフモデルを用いた生存曲線のシミュレーションと経済的価値の算出 [cite: 313, 315]
    """
    # 簡易生命表のベース死亡確率（q_x）のモックデータ（実際は厚労省の統計データ等を利用）
    base_q = 0.002
    
    survival_curve = []
    current_survival = 100.0
    ideal_survival = 100.0
    
    total_loss = 0
    total_gain = 0
    
    # シミュレーション期間: 現在年齢から85歳まで [cite: 1405]
    for current_age in range(user_data.age, 86, 5):
        # 年齢による死亡確率の上昇を加味
        age_factor = 1.0 + ((current_age - 40) * 0.05)
        
        # 補正死亡確率の算出: q_adj = 1 - (1 - q_base)^HR [cite: 353]
        q_current = 1 - (1 - (base_q * age_factor)) ** hr_current
        q_ideal = 1 - (1 - (base_q * age_factor)) ** hr_ideal
        
        # 生存曲線の記録
        survival_curve.append(SurvivalDataPoint(
            age=current_age,
            current=round(current_survival, 1),
            ideal=round(ideal_survival, 1)
        ))
        
        # 経済的価値の算出（65歳定年までの労働価値） [cite: 315]
        if current_age <= 65:
            # 5年間の生存確率の差分 × 平均年収を機会損失・獲得ボーナスとして計算 [cite: 129, 315]
            # ※本来は積分等の精密な計算を行いますが、API用に簡略化しています
            diff_from_base = (100.0 - current_survival) / 100.0
            diff_from_ideal = (ideal_survival - current_survival) / 100.0
            
            # 万円単位に変換
            yearly_income_man = user_data.annual_income / 10000 
            
            total_loss += int(diff_from_base * yearly_income_man * 5)
            total_gain += int(diff_from_ideal * yearly_income_man * 5)
        
        # 次のステップの生存確率を更新
        current_survival = current_survival * (1 - q_current * 5)
        ideal_survival = ideal_survival * (1 - q_ideal * 5)
        
        # 0未満にならないよう補正
        current_survival = max(0.0, current_survival)
        ideal_survival = max(0.0, ideal_survival)

    return survival_curve, EconomicMetrics(currentLoss=total_loss, potentialGain=total_gain)

@app.post("/api/v1/simulate", response_model=SimulationResponse)
def run_simulation(user_data: UserDataInput):
    """
    フロントエンドからの入力を受け取り、シミュレーション結果を返すエンドポイント
    """
    try:
        # 1. ハザード比の動的計算 [cite: 311]
        hr_current, hr_ideal = calculate_hazard_ratios(user_data)
        
        # 2. XAI（SHAP値）の計算 [cite: 317]
        shap_values = generate_shap_values(user_data)
        
        # 3. 生存曲線と経済的価値のシミュレーション [cite: 313, 315]
        survival_curve, economic_metrics = simulate_survival_and_economics(user_data, hr_current, hr_ideal)
        
        # 4. レスポンスの返却
        return SimulationResponse(
            survival_curve=survival_curve,
            shap_values=shap_values,
            economic_metrics=economic_metrics
        )
        
    except Exception as e:
        # エラーハンドリング
        raise HTTPException(status_code=500, detail=f"シミュレーション中にエラーが発生しました: {str(e)}")

2026年4月より、日本の胃がん検診は職域検診と対策型検診が統合され「組織型検診」へ移行する。対象年齢は50歳以上、間隔は2年に1回（隔年）へと厳格化される。
この「年齢のみを基準とした画一的（One-size-fits-all）な検診モデル」は、以下の深刻な問題を引き起こす。
* **医学的死角**: 50歳未満のピロリ菌感染者や、除菌後で高度萎縮性胃炎を残すハイリスク者の見逃し。
* **クリニックの経営危機**: 胃内視鏡検査の総件数が激減し、春日クリニックで66%、パークサイドクリニックで77%の収益減が試算される。
* **現場の疲弊**: 低リスク層への無作為な検査によるノーファインディング（徒労感）とプレッシャー。

**ミッション**: 「1: 受診者の医学的妥当性」「2: 職員の労働環境」「3: 検診業者の収益維持」というトリレンマを、AIによる「リスク層別化」と「高付加価値なプレミアム検診枠への誘導」によって同時に解決すること。

## 3. コア・アルゴリズム (Core Algorithms)
エージェントは以下のロジックに基づき、ユーザーの将来リスクと経済的価値をシミュレーションする。

1.  **動的ハザード補正 (Dynamic Hazard Ratio)**:
    * ベースラインに対する死亡確率のハザード比（HR）を計算。
    * ピロリ菌（現感染、除菌済）、萎縮性胃炎（軽度、高度）、喫煙歴による補正。
    * **特許要件（重要）**: 高齢（75歳以上）かつ低体重（BMI 18.5未満）の場合、フレイルリスクとみなしHRを動的にブースト（例：1.6 → 1.8）する。
2.  **フェイルセーフ機構 (Fail-safe)**:
    * ユーザーが「ピロリ菌感染歴」または「萎縮性胃炎の所見」を『不明』とした場合、不確実なスコアによる誤った安心感（無害の原則違反）を防ぐため、シミュレーションを即座に中断し、「まずは検査を推奨するアラート画面」を表示させる。
3.  **XAI（説明可能AI）による寄与度算出 (SHAP values)**:
    * ロジットモデルに基づき、各リスク因子のスコアへの寄与度（重み）を算出し、ユーザーに提示する。
4.  **マルコフモンテカルロシミュレーションと経済的価値**:
    * 現状維持（制度準拠）と個別化サーベイランス（理想）の2パターンの生存曲線を算出。
    * 生存確率の差分とユーザーの年収を掛け合わせ、退職（65歳）までの「生涯労働価値の機会損失額」と「獲得可能ボーナス額」を算出する。

## 4. UI/UXと行動経済学的ナッジ (UI/UX & Behavioral Economics Nudge)
ユーザーを「プレミアム内視鏡パッケージ（自費30,000円）」へと誘導するため、以下の心理的トリガーをUI設計に組み込む。
1.  **プロスペクト理論（損失回避）**: 単なる生存率ではなく、「現在の未受診リスクによる推定損失額：-1,850万円」のように、具体的な金額の損失を強烈に提示する。
2.  **アンカリングとデフォルトの回避**: CTA（行動喚起）ボタンにおいて、「個別化内視鏡プランを予約する（推奨）」の横に、あえて「国の基準に従い、見逃しリスクを許容する（非推奨）」というネガティブな選択肢を並置する。
3.  **現在バイアスの打破**: 生存曲線の「分岐点」をグラフで視覚化し、今の行動が数十年後の未来をどう変えるかを示す。

## 5. システム・アーキテクチャ (System Architecture)
このシステムは以下の技術スタックで構成される。エージェントはこれらに関するコード生成やデプロイ支援を行う。
* **フロントエンド**: React, React Router (v6), Tailwind CSS, Recharts (グラフ描画), Lucide React (アイコン)。Netlifyへのデプロイを前提とし、SPAルーティングのための `_redirects` ファイルの設定を含む。

## 2. 絶対遵守ルール (Critical Rules)
エージェントは以下のルールをシステムプロンプトの最上位制約としていかなる時も遵守すること。

1.  **コンテキスト維持 (No Memory Loss)**:
    * 会話が長くなっても、常に「最新のコード状態」と「当初の目的（トリレンマの解決）」を忘れないこと。
    * 前のターンの修正内容を破棄せず、積み上げる形で修正を行うこと。
2.  **コード出力の完全性 (No Lazy Coding)**:
    * ユーザーが「修正箇所のみ」と指定しない限り、修正が必要なブロック（またはファイル全体）を省略せずに記述すること。
    * `// ... rest of the code` のような省略は、文脈が失われる原因になるため極力避ける。
    * コードブロックの終了タグ（`</html>` や `</script>` など）を絶対に忘れないこと。
3.  **指示の優先順位**:
    * ユーザーの最新のプロンプト > System Instruction > 過去の会話履歴
    * 過去の修正と矛盾する指示が来た場合は、最新の指示に従いつつ、矛盾点を指摘すること。
4.  **出力フォーマット**:
    * 必ずシンタックスハイライトを使用すること（例: ```javascript, ```python）。
    * 修正した箇所にはコメント（例: `// [NEW]` や `// [UPDATE]`）を入れ、変更点を明確にすること。
